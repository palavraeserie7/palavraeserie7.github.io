create unique index if not exists editorial_contents_author_slug_idx on public.editorial_contents(author_id, slug); create index if not exists editorial_contents_category_idx on public.editorial_contents(category); create index if not exists editorial_contents_status_idx on public.editorial_contents(status); alter table public.profiles add column if not exists role text not null default 'user'; alter table public.editorial_contents enable row level security; create policy "owner can read own editorial contents" on public.editorial_contents for select to authenticated using (author_id = auth.uid() and exists (select 1 from public.profiles p where p.id = auth.uid() and p.role = 'owner')); create policy "public can read published free contents" on public.editorial_contents for select to authenticated using (status = 'Publicado' and access_level = 'Free'); create policy "owner can create editorial contents" on public.editorial_contents for insert to authenticated with check (author_id = auth.uid() and exists (select 1 from public.profiles p where p.id = auth.uid() and p.role = 'owner')); create policy "owner can update editorial contents" on public.editorial_contents for update to authenticated using (author_id = auth.uid() and exists (select 1 from public.profiles p where p.id = auth.uid() and p.role = 'owner')) with check (author_id = auth.uid() and exists (select 1 from public.profiles p where p.id = auth.uid() and p.role = 'owner')); create policy "owner can delete editorial contents" on public.editorial_contents for delete to authenticated using (author_id = auth.uid() and exists (select 1 from public.profiles p where p.id = auth.uid() and p.role = 'owner')); create or replace function public.touch_editorial_contents_updated_at() returns trigger language plpgsql as $$ begin new.updated_at = now(); return new; end; $$; drop trigger if exists editorial_contents_touch on public.editorial_contents;
